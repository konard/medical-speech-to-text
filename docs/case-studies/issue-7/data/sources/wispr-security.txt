  function removeRogueQuotes() {

<script>
  function initCtaDots() {
    const ctaCount = 100;
    const ctaDuration = 20;
    const pathOpts = {
      path: '#curve-cta',
      align: '#curve-cta',
      alignOrigin: [0.5, 0.5],
    };

    const curveElement = document.querySelector('#curve-cta');
    if (!curveElement) {
      return;
    }

    const svgWrapper = curveElement.closest('svg').parentNode;
    svgWrapper.style.position = 'relative';

    for (let i = 0; i < ctaCount; i++) {
      const dot = document.createElement('div');
      dot.classList.add('cta_dot');
      gsap.set(dot, { position: 'absolute' });
      svgWrapper.appendChild(dot);

      const tween = gsap.to(dot, {
        duration: ctaDuration,
        repeat: -1,
        ease: 'none',
        motionPath: pathOpts,
        immediateRender: false,
      });
      tween.progress(i / ctaCount);
    }
  }

  // create a GSAP matchMedia instance
  const mm = gsap.matchMedia();

  // add a rule for screens ≥768px
  mm.add('(min-width: 768px)', () => {
    // when this rule first matches, wait for load + a frame then init
    window.addEventListener('load', () => {
      requestAnimationFrame(initCtaDots);
    });

    // return a cleanup function to remove the listener if rule no longer matches
    return () => {
      window.removeEventListener('load', initCtaDots);
    };
  });
</script>

<script>
  (function () {
    var SCROLL_OFFSET = 0.25;
    var SCROLL_DURATION = 300;
    var currentAnimId = 0;

    function easeInOutQuad(t) {
      return t < 0.5 ? 2 * t * t : 1 - Math.pow(-2 * t + 2, 2) / 2;
    }

    function smoothScrollTo(targetY, duration) {
      var animId = ++currentAnimId;
      var startY = window.pageYOffset;
      var diff = targetY - startY;
      var startTime = null;

      function step(timestamp) {
        if (animId !== currentAnimId) return;
        if (!startTime) startTime = timestamp;
        var progress = Math.min((timestamp - startTime) / duration, 1);
        window.scrollTo({ top: startY + diff * easeInOutQuad(progress), behavior: 'instant' });
        if (progress < 1) requestAnimationFrame(step);
      }
      requestAnimationFrame(step);
    }

    function initInstance(root) {
      var navItems = Array.prototype.slice.call(root.querySelectorAll('[data-tab]'));
      var indicator = root.querySelector('[data-tab-indicator]');
      var panels = navItems.map(function (item) {
        var target = item.getAttribute('data-tab');
        return root.querySelector('[data-tab-target="' + target + '"]');
      });

      function moveIndicator(item) {
        if (!indicator || !item) return;
        var label = item.querySelector('[data-tab-label]') || item;
        indicator.style.transform = 'translateY(' + label.offsetTop + 'px)';
        indicator.style.height = label.offsetHeight + 'px';
      }

      function setActive(i) {
        navItems.forEach(function (item, idx) {
          item.classList.toggle('is-active', idx === i);
        });
        moveIndicator(navItems[i]);
      }

      navItems.forEach(function (item, i) {
        item.addEventListener('click', function (e) {
          e.preventDefault();
          e.stopPropagation();
          if (!panels[i]) return;
          var rect = panels[i].getBoundingClientRect();
          var targetY = window.pageYOffset + rect.top - window.innerHeight * SCROLL_OFFSET;
          smoothScrollTo(targetY, SCROLL_DURATION);
        });
      });

      if ('IntersectionObserver' in window) {
        var io = new IntersectionObserver(
          function (entries) {
            entries.forEach(function (e) {
              if (e.isIntersecting) {
                var idx = panels.indexOf(e.target);
                if (idx > -1) setActive(idx);
              }
            });
          },
          { rootMargin: '-40% 0px -50% 0px' },
        );
        panels.forEach(function (p) {
          if (p) io.observe(p);
        });
      }

      setActive(0);
    }

    document.addEventListener('DOMContentLoaded', function () {
      document.querySelectorAll('[data-tab-root]').forEach(initInstance);
    });
  })();
</script></body></html>
